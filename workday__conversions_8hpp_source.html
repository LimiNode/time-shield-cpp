72 TIME_SHIELD_CONSTEXPR inline int workday_index_in_month(year_t year, int month, int day) noexcept {
73 if (!is_workday(year, month, day)) {
74 return 0;
75 }
76 const int days = num_days_in_month(year, month);
77 if (days <= 0) {
78 return 0;
79 }
80 int index = 0;
81 for (int current = 1; current <= days; ++current) {
82 if (is_workday(year, month, current)) {
83 ++index;
84 if (current == day) {
85 return index;
86 }
87 }
88 }
89 return 0;
90 }

105 TIME_SHIELD_CONSTEXPR inline bool is_within_first_workdays_of_month(year_t year, int month, int day, int count) noexcept {
106 if (count <= 0) {
