659 static bool try_parse_iso8601_buffer(const char* data, std::size_t size, DateTime& out) noexcept {
660 if (data == nullptr) {
661 return false;
662 }
663 DateTimeStruct dt = create_date_time_struct(0);
664 TimeZoneStruct tz = create_time_zone_struct(0, 0, true);
665 if (!time_shield::parse_iso8601(data, size, dt, tz)) {
666 return false;
667 }
668 const tz_t offset = time_zone_struct_to_offset(tz);
669 if (!is_valid_tz_offset(offset)) {
670 return false;
671 }
672 const ts_ms_t utc_ms = dt_to_timestamp_ms(dt) - offset_to_ms(offset);
673 out = from_unix_ms(utc_ms, offset);
674 return true;
675 }
676
677 static DateTime parse_iso8601_buffer(const char* data, std::size_t size) {
678 DateTime result;
679 if (!try_parse_iso8601_buffer(data, size, result)) {
680 throw std::invalid_argument("Invalid ISO8601 datetime");
681 }
682 return result;
683 }
