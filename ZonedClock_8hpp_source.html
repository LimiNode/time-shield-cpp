109 bool try_set_zone(const std::string& zone_spec) noexcept {
110 const std::string trimmed = trim_ascii(zone_spec);
111 if (trimmed.empty()) {
112 return false;
113 }
114
115 TimeZone parsed_zone = UNKNOWN;
116 if (parse_time_zone_name(trimmed, parsed_zone)) {
117 set_zone(parsed_zone);
118 return true;
119 }
120
121 TimeZoneStruct parsed_offset = create_time_zone_struct(0, 0, true);
122 if (!parse_time_zone(trimmed, parsed_offset)) {
123 return false;
124 }
125
126 return try_set_offset(time_zone_struct_to_offset(parsed_offset));
127 }

192 LocalTimeResolution resolve_local_time_ms(ts_ms_t local_ms) const noexcept {
193 if (local_ms == ERROR_TIMESTAMP) {
194 LocalTimeResolution result = {
195 LocalTimeStatus::unsupported,
196 ERROR_TIMESTAMP,
197 ERROR_TIMESTAMP

208 time_shield::to_utc_ms(local_ms, m_offset),

324 static std::string trim_ascii(const std::string& value) {
325 std::size_t begin = 0;
326 std::size_t end = value.size();
327 while (begin < end && is_ascii_space(value[begin])) {
328 ++begin;
329 }
330 while (end > begin && is_ascii_space(value[end - 1])) {
331 --end;
332 }
333 return value.substr(begin, end - begin);
334 }
335
336 static bool is_ascii_space(char ch) noexcept {
