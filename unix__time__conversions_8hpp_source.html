24 template<class T = year_t>
25 TIME_SHIELD_CONSTEXPR T years_since_epoch(ts_t ts) noexcept {
26 // 9223372029693630000 reaches year 292277024400 from the 2000 epoch.
27 // This value overflows the n_400_years * SEC_PER_400_YEARS calculation.
28 // The supported bound is reduced to 9223371890843040000.
29 constexpr int64_t BIAS_292277022000 = 9223371890843040000LL;
30 constexpr int64_t BIAS_2000 = 946684800LL;
31
32 int64_t y = MAX_YEAR;
33 int64_t secs = -((ts - BIAS_2000) - BIAS_292277022000);
34
35 const int64_t n_400_years = secs / SEC_PER_400_YEARS;
36 secs -= n_400_years * SEC_PER_400_YEARS;
37 y -= n_400_years * 400;
38
39 const int64_t n_100_years = secs / SEC_PER_100_YEARS;
40 secs -= n_100_years * SEC_PER_100_YEARS;
41 y -= n_100_years * 100;
42
43 const int64_t n_4_years = secs / SEC_PER_4_YEARS;
44 secs -= n_4_years * SEC_PER_4_YEARS;
45 y -= n_4_years * 4;
46
47 const int64_t n_1_years = secs / SEC_PER_YEAR;
48 secs -= n_1_years * SEC_PER_YEAR;
49 y -= n_1_years;
50
51 y = secs == 0 ? y : y - 1;
52 return y - UNIX_EPOCH;
53 }

85 template<class Year, class Month, class Day>
86 TIME_SHIELD_CONSTEXPR inline dse_t date_to_unix_day(
87 Year year,
88 Month month,
89 Day day) noexcept {
90 const int64_t y = static_cast<int64_t>(year) - (static_cast<int64_t>(month) <= 2 ? 1 : 0);
91 const int64_t m = static_cast<int64_t>(month) <= 2
92 ? static_cast<int64_t>(month) + 9
93 : static_cast<int64_t>(month) - 3;
94 const int64_t era = (y >= 0 ? y : y - 399) / 400;
95 const int64_t yoe = y - era * 400;
96 const int64_t doy = (153 * m + 2) / 5 + static_cast<int64_t>(day) - 1;
97 const int64_t doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
98 return static_cast<dse_t>(era * 146097 + doe - 719468);
99 }
