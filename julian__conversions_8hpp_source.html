26 inline jd_t gregorian_dmy_to_jd_unchecked(double day, int64_t month, int64_t year) noexcept {
27 if (month == 1 || month == 2) {
28 year -= 1;
29 month += 12;
30 }
31 const double a = std::floor(static_cast<double>(year) / 100.0);
32 const double b = 2.0 - a + std::floor(a / 4.0);
33 const double jd = std::floor(365.25 * (static_cast<double>(year) + 4716.0))
34 + std::floor(30.6000001 * (static_cast<double>(month) + 1.0))
35 + day + b - 1524.5;
36 return static_cast<jd_t>(jd);
37 }

39 inline jdn_t gregorian_dmy_to_jdn_unchecked(int64_t day, int64_t month, int64_t year) noexcept {
40 const int64_t a = (14LL - month) / 12LL;
41 const int64_t y = year + 4800LL - a;
42 const int64_t m = month + 12LL * a - 3LL;
43 const int64_t jdn = day
44 + (153LL * m + 2LL) / 5LL
45 + 365LL * y
46 + y / 4LL
47 - y / 100LL
48 + y / 400LL
49 - 32045LL;
50 return static_cast<jdn_t>(jdn);
51 }
