305 inline bool try_parse_positive_int64(const char* data, std::size_t length, int64_t& value) noexcept {
306 value = 0;
307 if (data == nullptr || length == 0) {
308 return false;
309 }
310
311 const int64_t max_value = (std::numeric_limits<int64_t>::max)();
312 for (std::size_t i = 0; i < length; ++i) {
313 if (!is_ascii_digit(data[i])) {
314 return false;
315 }
316
317 const int digit = data[i] - '0';
318 if (value > (max_value - digit) / 10) {
319 value = 0;
320 return false;
321 }
322 value = value * 10 + digit;
323 }
324
325 return value > 0;
326 }

418 seconds = 0;
419 if (data == nullptr || length == 0) {
420 return false;
421 }
422
423 const char* begin = data;
424 const char* end = data + length;
425 while (begin < end && is_ascii_space(*begin)) {
426 ++begin;
427 }
428 while (end > begin && is_ascii_space(*(end - 1))) {
429 --end;
430 }
431
432 if (begin == end) {
433 return false;
434 }
435
436 int64_t multiplier = 1;
437 int64_t unit_seconds = 0;
438 int64_t result = 0;

440 if (is_ascii_digit(*begin)) {

442 while (cursor < end && is_ascii_digit(*cursor)) {
443 ++cursor;
444 }
445

560 if (p >= end || !is_ascii_digit(*p)) {
561 return false;

861 inline bool parse_time_zone(const char* data, std::size_t length, TimeZoneStruct& tz) noexcept {
862 if (!data) {
863 return false;
864 }
865
866 if (length == 0) {
867 tz.hour = 0;
868 tz.min = 0;
869 tz.is_positive = true;
870 return true;
871 }
872
873 if (length == 1 && (data[0] == 'Z' || data[0] == 'z')) {
874 tz.hour = 0;
875 tz.min = 0;
876 tz.is_positive = true;
877 return true;
878 }
879
880 if (length != 6) {
881 return false;
882 }
883
884 const char sign = data[0];
885 if (sign != '+' && sign != '-') {
886 return false;
887 }
888 if (data[3] != ':') {
889 return false;
890 }
891 if (!detail::is_ascii_digit(data[1]) || !detail::is_ascii_digit(data[2]) ||
892 !detail::is_ascii_digit(data[4]) || !detail::is_ascii_digit(data[5])) {
893 return false;
894 }
895
896 tz.is_positive = (sign == '+');
897 tz.hour = (data[1] - '0') * 10 + (data[2] - '0');
898 tz.min = (data[4] - '0') * 10 + (data[5] - '0');
899
900 return is_valid_time_zone(tz);
901 }

989// ISO8601 parsing (C-style, no regex, no allocations)
990//------------------------------------------------------------------------------
991
1011 inline bool parse_iso8601(const char* input, std::size_t length,
1012 DateTimeStruct& dt, TimeZoneStruct& tz) noexcept {
1013 if (!input) {
1014 return false;
1015 }
1016
1017 const char* p = input;
1018 const char* end = input + length;
1019
1020 detail::skip_spaces(p, end);
1021
1022 dt = create_date_time_struct(0);
1023 tz = create_time_zone_struct(0, 0);
1024 tz.is_positive = true;
1025
1026 const char* const date_start = p;
1027 const char* date_end = p;
1028 while (date_end < end && *date_end != 'T' && *date_end != 't' && !detail::is_ascii_space(*date_end)) {
1029 ++date_end;
1030 }
1031
1032 bool parsed_iso_week_date = false;
1033 if (date_end > date_start) {
1034 IsoWeekDateStruct iso_date{};
1035 if (parse_iso_week_date(date_start, static_cast<std::size_t>(date_end - date_start), iso_date)) {
1036 const DateStruct calendar_date = iso_week_date_to_date(iso_date);
1037 dt.year = calendar_date.year;
1038 dt.mon = calendar_date.mon;
1039 dt.day = calendar_date.day;
1040 p = date_end;
1041 parsed_iso_week_date = true;
1042 }
1043 }
1044
1045 if (!parsed_iso_week_date) {
1046 // ---- Date: YYYY<sep>MM<sep>DD
1047 if (!detail::parse_4digits_year(p, end, dt.year)) {
1048 return false;
1049 }
1050 if (p >= end) {
1051 return false;
1052 }
1053 const char sep1 = *p;
1054 if (sep1 != '-' && sep1 != '/' && sep1 != '.') {
1055 return false;
1056 }
1057 ++p;
1058
1059 if (!detail::parse_2digits(p, end, dt.mon)) {
1060 return false;
1061 }
1062 if (p >= end) {
1063 return false;
1064 }
1065 const char sep2 = *p;
1066 if (sep2 != '-' && sep2 != '/' && sep2 != '.') {
1067 return false;
1068 }
1069 ++p;
1070
1071 if (!detail::parse_2digits(p, end, dt.day)) {
1072 return false;
1073 }
1074 }
1075
1076 if (!is_valid_date(dt.year, dt.mon, dt.day)) {
1077 return false;
1078 }
1079
1080 // Date-only?
1081 {
1082 const char* q = p;
1083 detail::skip_spaces(q, end);
1084 if (q == end) {
1085 // dt already has time=0 ms=0
1086 return is_valid_date_time(dt);
1087 }
1088 }
1089
1090 // ---- Date/time separator: 'T' or whitespace
1091 if (p >= end) {
1092 return false;
1093 }
1094
1095 if (*p == 'T' || *p == 't') {
1096 ++p;
1097 } else
1098 if (detail::is_ascii_space(*p)) {
1099 // allow one or more spaces
1100 detail::skip_spaces(p, end);
1101 } else {
1102 return false;
1103 }
1104
1105 // ---- Time: hh:mm[:ss][.frac]
1106 if (!detail::parse_2digits(p, end, dt.hour)) {
1107 return false;
1108 }
1109 if (p >= end || *p != ':') {
1110 return false;
1111 }
1112 ++p;
1113
1114 if (!detail::parse_2digits(p, end, dt.min)) {
1115 return false;
1116 }
1117
1118 dt.sec = 0;
1119 dt.ms = 0;
1120 bool has_seconds = false;
1121
1122 // Optional :ss
1123 if (p < end && *p == ':') {
1124 ++p;
1125 if (!detail::parse_2digits(p, end, dt.sec)) {
1126 return false;
1127 }
1128 has_seconds = true;
1129 }
1130
1131 // Optional .fraction (allowed only if we had seconds in original regex,
1132 // but we accept it when seconds are present; for hh:mm (no seconds) we keep it strict).
1133 if (p < end && *p == '.') {
1134 // require seconds field to exist (avoid accepting YYYY-MM-DDThh:mm.xxx)
1135 if (!has_seconds) {
1136 // Ambiguous: could be "hh:mm.fff" which is not in your original formats.
1137 // Keep strict to preserve behavior.
1138 return false;
1139 }
1140
1141 ++p;
1142 int ms = 0;
1143 if (!detail::parse_fraction_to_ms(p, end, ms)) {
1144 return false;
1145 }
1146 dt.ms = ms;
1147 }
1148
1149 // ---- Optional timezone: [spaces] (Z | +/-HH:MM)
1150 detail::skip_spaces(p, end);
1151
1152 if (p < end) {
1153 if (*p == 'Z' || *p == 'z') {
1154 tz.hour = 0;
1155 tz.min = 0;
1156 tz.is_positive = true;
1157 ++p;
1158 } else if (*p == '+' || *p == '-') {
1159 // need 6 chars
1160 if (static_cast<std::size_t>(end - p) < 6) {
1161 return false;
1162 }
1163 if (!parse_time_zone(p, 6, tz)) {
1164 return false;
1165 }
1166 p += 6;
1167 }
1168 }
1169
1170 detail::skip_spaces(p, end);
1171 if (p != end) {
1172 return false;
1173 }
1174
1175 return is_valid_date_time(dt);
1176 }

1705 inline bool sec_of_day(const std::string& str, T& sec) {
1706 if (str.empty()) return false;
1707
1708 const char* p = str.c_str();
1709 int parts[3] = {0, 0, 0}; // hour, minute, second
1710 int idx = 0;
1711
1712 while (*p && idx < 3) {
1713 // Parse integer
1714 int value = 0;
1715 bool has_digit = false;
1716
1717 while (*p >= '0' && *p <= '9') {
1718 has_digit = true;
1719 value = value * 10 + (*p - '0');
1720 ++p;
1721 }
1722
1723 if (!has_digit) return false;
1724 parts[idx++] = value;
1725
1726 // Expect colon or end
1727 if (*p == ':') {
1728 ++p;
1729 } else if (*p == '\0') {
1730 break;
1731 } else {
1732 return false; // unexpected character
1733 }
1734 }
1735
1736 if (idx == 0) return false;
1737 if (!is_valid_time(parts[0], parts[1], parts[2])) return false;
1738
1739 sec = static_cast<T>(sec_of_day(parts[0], parts[1], parts[2]));
1740 return true;
1741 }
