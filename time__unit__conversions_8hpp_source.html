150 template<class T1 = ts_ms_t, class T2>
151 TIME_SHIELD_CONSTEXPR T1 sec_to_ms(T2 ts) noexcept {
152# ifdef TIME_SHIELD_CPP17
153 if constexpr (std::is_floating_point_v<T2>) {
154 return static_cast<T1>(std::round(ts * static_cast<T2>(MS_PER_SEC)));
155 } else {
156 return static_cast<T1>(ts) * static_cast<T1>(MS_PER_SEC);
157 }
158# else
159 return static_cast<T1>(sec_to_ms_impl(ts, typename std::conditional<
160 (std::is_same<T2, double>::value || std::is_same<T2, float>::value),
161 std::true_type,
162 std::false_type
163 >::type{}));
164# endif
165 }
