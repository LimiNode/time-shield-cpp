48 TIME_SHIELD_CONSTEXPR inline tz_t tz_offset_hm(int hour, int min = 0) noexcept {
49 const int sign = (hour < 0) ? -1 : 1;
50 const int64_t ah = (hour < 0) ? -static_cast<int64_t>(hour) : static_cast<int64_t>(hour);
