90 template<class Rep, class Period>
91 void start(std::chrono::duration<Rep, Period> timeout) noexcept {
92 const time_point now = clock::now();
93 if (timeout <= decltype(timeout)::zero()) {
94 start(now);
95 return;
96 }
97
98 duration safe_duration = std::chrono::duration_cast<duration>(timeout);
99 if (safe_duration <= duration::zero()) {
100 safe_duration = duration(1);
101 }
102
103 const time_point max_time = (time_point::max)();
104 const duration max_offset = max_time - now;
105 if (safe_duration >= max_offset) {
106 start(max_time);
107 return;
108 }
109
110 start(now + safe_duration);
111 }

208 void add(duration extend_by) noexcept {
209 if (!m_is_running || extend_by <= duration::zero()) {
210 return;
211 }
212
213 const time_point now = clock::now();
214 const time_point base = m_deadline > now ? m_deadline : now;
215 const duration max_offset = (time_point::max)() - base;
216 const duration safe_offset = extend_by < max_offset ? extend_by : max_offset;
217 m_deadline = base + safe_offset;
218 }
