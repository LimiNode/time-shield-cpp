54 inline ts_t cet_to_gmt_impl(ts_t cet) {
55 DateTimeStruct dt = to_date_time(cet);
56 int max_days = num_days_in_month(dt.year, dt.mon);
57 const int OLD_START_SUMMER_HOUR = 2;
58 const int OLD_STOP_SUMMER_HOUR = 3;
59 const int NEW_SUMMER_HOUR = 1;
60
61 if(dt.year < 2002) {
62 if(dt.mon > MAR && dt.mon < OCT) {
63 return cet - SEC_PER_HOUR * 2;
64 } else if(dt.mon == MAR) {
65 for(int d = max_days; d >= dt.day; --d) {
66 if(day_of_week_date(dt.year, MAR, d) == SUN) {
67 if(d == dt.day) {
68 if(dt.hour >= OLD_START_SUMMER_HOUR) {
69 return cet - SEC_PER_HOUR * 2;
70 }
71 return cet - SEC_PER_HOUR;
72 }
73 return cet - SEC_PER_HOUR;
74 }
75 }
76 return cet - SEC_PER_HOUR * 2;
77 } else if(dt.mon == OCT) {
78 for(int d = max_days; d >= dt.day; --d) {
79 if(day_of_week_date(dt.year, OCT, d) == SUN) {
80 if(d == dt.day) {
81 if(dt.hour >= OLD_STOP_SUMMER_HOUR) {
82 return cet - SEC_PER_HOUR;
83 }
84 return cet - SEC_PER_HOUR;
85 }
86 return cet - SEC_PER_HOUR * 2;
87 }
88 }
89 return cet - SEC_PER_HOUR;
90 }
91 return cet - SEC_PER_HOUR;
92 }
93
94 if(dt.mon > MAR && dt.mon < OCT) {
95 return cet - SEC_PER_HOUR * 2;
96 }
97 if(dt.mon == MAR) {
98 for(int d = max_days; d >= dt.day; --d) {
99 if(day_of_week_date(dt.year, MAR, d) == SUN) {
100 if(d == dt.day) {
101 if(dt.hour >= (NEW_SUMMER_HOUR + 2)) {
102 return cet - SEC_PER_HOUR * 2;
103 }
104 return cet - SEC_PER_HOUR;
105 }
106 return cet - SEC_PER_HOUR;
107 }
108 }
109 return cet - SEC_PER_HOUR * 2;
110 }
111 if(dt.mon == OCT) {
112 for(int d = max_days; d >= dt.day; --d) {
113 if(day_of_week_date(dt.year, OCT, d) == SUN) {
114 if(d == dt.day) {
115 if(dt.hour >= (NEW_SUMMER_HOUR + 1)) {
116 return cet - SEC_PER_HOUR;
117 }
118 return cet - SEC_PER_HOUR * 2;
119 }
120 return cet - SEC_PER_HOUR * 2;
121 }
122 }
123 }
124 return cet - SEC_PER_HOUR;
125 }

127 inline ts_t gmt_to_cet_impl(ts_t gmt) {
128 DateTimeStruct dt = to_date_time(gmt);
129 const int SWITCH_HOUR = 1;
130
131 if(dt.mon > MAR && dt.mon < OCT) {
132 return gmt + SEC_PER_HOUR * 2;
133 }
134 if(dt.mon == MAR) {
135 int last = last_sunday_month_day(dt.year, MAR);
136 if(dt.day > last) {
137 return gmt + SEC_PER_HOUR * 2;
138 }
139 if(dt.day < last) {
140 return gmt + SEC_PER_HOUR;
141 }
142 if(dt.hour >= SWITCH_HOUR) {
143 return gmt + SEC_PER_HOUR * 2;
144 }
145 return gmt + SEC_PER_HOUR;
146 }
147 if(dt.mon == OCT) {
148 int last = last_sunday_month_day(dt.year, OCT);
149 if(dt.day > last) {
150 return gmt + SEC_PER_HOUR;
151 }
152 if(dt.day < last) {
153 return gmt + SEC_PER_HOUR * 2;
154 }
155 if(dt.hour >= SWITCH_HOUR) {
156 return gmt + SEC_PER_HOUR;
157 }
158 return gmt + SEC_PER_HOUR * 2;
159 }
160 return gmt + SEC_PER_HOUR;
161 }

171 inline bool is_us_eastern_dst_local(const DateTimeStruct& dt) {
172 const int SWITCH_HOUR = 2;
173 int start_day = 0;
174 int end_day = 0;
175 int start_month = 0;
176 int end_month = 0;
177
178 if(dt.year >= 2007) {
179 start_month = MAR;
180 end_month = NOV;
181 int first_sunday_march = static_cast<int>(
182 1 + (DAYS_PER_WEEK - day_of_week_date(dt.year, MAR, 1)) % DAYS_PER_WEEK);
183 start_day = first_sunday_march + 7;
184 end_day = static_cast<int>(
185 1 + (DAYS_PER_WEEK - day_of_week_date(dt.year, NOV, 1)) % DAYS_PER_WEEK);
186 } else {
187 start_month = APR;
188 end_month = OCT;
189 start_day = static_cast<int>(
190 1 + (DAYS_PER_WEEK - day_of_week_date(dt.year, APR, 1)) % DAYS_PER_WEEK);
191 end_day = last_sunday_month_day(dt.year, OCT);
192 }
193
194 if(dt.mon > start_month && dt.mon < end_month) {
195 return true;
196 }
197 if(dt.mon < start_month || dt.mon > end_month) {
198 return false;
199 }
200 if(dt.mon == start_month) {
201 if(dt.day > start_day) {
202 return true;
203 }
204 if(dt.day < start_day) {
205 return false;
206 }
207 return dt.hour >= SWITCH_HOUR;
208 }
209 if(dt.mon == end_month) {
210 if(dt.day < end_day) {
211 return true;
212 }
213 if(dt.day > end_day) {
214 return false;
215 }
216 return dt.hour < SWITCH_HOUR;
217 }
218 return false;
219 }

221 inline bool fixed_zone_offset(TimeZone zone, tz_t& utc_offset) {
222 switch(zone) {
223 case GMT:
224 case UTC:
225 case WET:
226 utc_offset = 0;
227 return true;
228 case WEST:
229 utc_offset = static_cast<tz_t>(SEC_PER_HOUR);
230 return true;
231 case CET:
232 utc_offset = static_cast<tz_t>(SEC_PER_HOUR);
233 return true;
234 case CEST:
235 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 2);
236 return true;
237 case EET:
238 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 2);
239 return true;
240 case EEST:
241 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 3);
242 return true;
243 case IST:
244 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 5 + SEC_PER_MIN * 30);
245 return true;
246 case MYT:
247 case WITA:
248 case SGT:
249 case PHT:
250 case HKT:
251 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 8);
252 return true;
253 case WIB:
254 case ICT:
255 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 7);
256 return true;
257 case WIT:
258 case JST:
259 case KST:
260 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 9);
261 return true;
262 case KZT:
263 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 5);
264 return true;
265 case TRT:
266 case BYT:
267 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 3);
268 return true;
269 case GST:
270 utc_offset = static_cast<tz_t>(SEC_PER_HOUR * 4);
271 return true;
272 default:
273 utc_offset = 0;
274 return false;
275 }

278 inline ts_ms_t zone_to_gmt_ms_by_seconds(ts_ms_t local_ms, TimeZone zone) {
279 if(local_ms == ERROR_TIMESTAMP) {
280 return ERROR_TIMESTAMP;
281 }
282 const ts_t local_sec = ms_to_sec<ts_t>(local_ms);
283 const ts_ms_t remainder_ms = local_ms - sec_to_ms<ts_ms_t>(local_sec);
284 const ts_t gmt_sec = zone_to_gmt(local_sec, zone);
285 if(gmt_sec == ERROR_TIMESTAMP) {
286 return ERROR_TIMESTAMP;
287 }
288 return sec_to_ms<ts_ms_t>(gmt_sec) + remainder_ms;
289 }

291 inline ts_ms_t gmt_to_zone_ms_by_seconds(ts_ms_t gmt_ms, TimeZone zone) {
292 if(gmt_ms == ERROR_TIMESTAMP) {
293 return ERROR_TIMESTAMP;
294 }
295 const ts_t gmt_sec = ms_to_sec<ts_t>(gmt_ms);
296 const ts_ms_t remainder_ms = gmt_ms - sec_to_ms<ts_ms_t>(gmt_sec);
297 const ts_t local_sec = gmt_to_zone(gmt_sec, zone);
298 if(local_sec == ERROR_TIMESTAMP) {
299 return ERROR_TIMESTAMP;
300 }

392 inline ts_t zone_to_gmt(ts_t local, TimeZone zone) {
393 if(local == ERROR_TIMESTAMP) {
394 return ERROR_TIMESTAMP;
395 }
396
397 tz_t utc_offset = 0;
398 switch(zone) {
399 case GMT:
400 case UTC:
401 return local;
402 case WET:
403 return detail::european_local_to_gmt(local, 0);
404 case CET:
405 return cet_to_gmt(local);
406 case EET:
407 return eet_to_gmt(local);
408 case WEST:
409 case CEST:
410 case EEST:
411 detail::fixed_zone_offset(zone, utc_offset);
412 return to_utc(local, utc_offset);
413 case ET:
414 return et_to_gmt(local);
415 case CT:
416 return ct_to_gmt(local);
417 case UNKNOWN:
418 return ERROR_TIMESTAMP;
419 default:
420 if(detail::fixed_zone_offset(zone, utc_offset)) {
421 return to_utc(local, utc_offset);
422 }
423 return ERROR_TIMESTAMP;
424 }

624 inline bool european_dst_at_utc_ms(ts_ms_t utc_ms) {
625 const DateTimeStruct dt = to_date_time(ms_to_sec<ts_t>(utc_ms));
626 const int start_day = last_sunday_month_day(dt.year, MAR);
627 const int end_day = last_sunday_month_day(dt.year, OCT);
628 const ts_ms_t start_ms =
629 to_timestamp_ms(dt.year, int(MAR), start_day, 1, 0, 0);
630 const ts_ms_t end_ms =
631 to_timestamp_ms(dt.year, int(OCT), end_day, 1, 0, 0);
632 return utc_ms >= start_ms && utc_ms < end_ms;
633 }

638 DAYS_PER_WEEK);

641 inline bool us_dst_at_utc_ms(TimeZone zone, ts_ms_t utc_ms) {
642 const DateTimeStruct dt = to_date_time(ms_to_sec<ts_t>(utc_ms));
643 const int year = static_cast<int>(dt.year);
644 int start_month = MAR;
645 int end_month = NOV;
646 int start_day = first_sunday_month_day(year, MAR) + 7;
647 int end_day = first_sunday_month_day(year, NOV);
648
649 if(dt.year < 2007) {
650 start_month = APR;
651 end_month = OCT;
652 start_day = first_sunday_month_day(year, APR);
653 end_day = last_sunday_month_day(year, OCT);
654 }
655
656 const tz_t standard_offset = dynamic_zone_standard_offset(zone);
657 const tz_t daylight_offset =
658 static_cast<tz_t>(standard_offset + SEC_PER_HOUR);
659 const ts_ms_t start_local =
660 to_timestamp_ms(dt.year, start_month, start_day, 2, 0, 0);
661 const ts_ms_t end_local =
662 to_timestamp_ms(dt.year, end_month, end_day, 2, 0, 0);
663 const ts_ms_t start_utc = to_utc_ms(start_local, standard_offset);
664 const ts_ms_t end_utc = to_utc_ms(end_local, daylight_offset);
665 return utc_ms >= start_utc && utc_ms < end_utc;
666 }

668 inline bool dynamic_offset_applies_at_utc_ms(TimeZone zone,
669 ts_ms_t utc_ms,
670 tz_t offset) {
671 if(is_european_dynamic_zone(zone)) {
672 const tz_t standard_offset = dynamic_zone_standard_offset(zone);
673 const tz_t expected =
674 static_cast<tz_t>(standard_offset +
675 (european_dst_at_utc_ms(utc_ms)
676 ? SEC_PER_HOUR
677 : 0));
678 return offset == expected;
679 }
680
681 if(is_us_dynamic_zone(zone)) {
682 const tz_t standard_offset = dynamic_zone_standard_offset(zone);
683 const tz_t expected =
684 static_cast<tz_t>(standard_offset +
685 (us_dst_at_utc_ms(zone, utc_ms)
686 ? SEC_PER_HOUR
687 : 0));
688 return offset == expected;
689 }
690
691 return false;
692 }

694 inline void add_local_time_candidate(LocalTimeResolution& result,
695 ts_ms_t local_ms,
696 TimeZone zone,
697 tz_t offset) {
698 const ts_ms_t candidate = to_utc_ms(local_ms, offset);
699 if(candidate == ERROR_TIMESTAMP ||
700 to_local_ms(candidate, offset) != local_ms ||
701 !dynamic_offset_applies_at_utc_ms(zone, candidate, offset) ||
702 result.first_utc_ms == candidate ||
703 result.second_utc_ms == candidate) {
704 return;
705 }
706
707 if(result.first_utc_ms == ERROR_TIMESTAMP) {
708 result.first_utc_ms = candidate;
709 return;
710 }
711
712 if(result.second_utc_ms == ERROR_TIMESTAMP) {
713 result.second_utc_ms = candidate;
714 }
715 }

717 inline LocalTimeResolution resolve_with_dynamic_offsets(
718 ts_ms_t local_ms,
719 TimeZone zone,
720 tz_t first_offset,
721 tz_t second_offset) {
722 LocalTimeResolution result =
723 make_local_time_resolution(LocalTimeStatus::nonexistent);
724
725 add_local_time_candidate(result, local_ms, zone, first_offset);
726 add_local_time_candidate(result, local_ms, zone, second_offset);
727
728 if(result.first_utc_ms == ERROR_TIMESTAMP) {
729 return result;
730 }
731
732 if(result.second_utc_ms == ERROR_TIMESTAMP) {
733 result.status = LocalTimeStatus::valid;
734 return result;
735 }
736
737 if(result.second_utc_ms < result.first_utc_ms) {
738 const ts_ms_t tmp = result.first_utc_ms;
739 result.first_utc_ms = result.second_utc_ms;
740 result.second_utc_ms = tmp;
741 }
742
743 result.status = LocalTimeStatus::ambiguous;
744 return result;
745 }

781 inline bool zone_offset_at_utc_ms(ts_ms_t utc_ms,
782 TimeZone zone,
783 tz_t& out) noexcept {
784 if(utc_ms == ERROR_TIMESTAMP || zone == UNKNOWN) {
785 return false;
786 }
787
788 if(zone == GMT || zone == UTC) {
789 out = 0;

806 ? SEC_PER_HOUR
807 : 0));
808 return true;
809 }

811 tz_t utc_offset = 0;
812 if(detail::fixed_zone_offset(zone, utc_offset)) {
813 out = utc_offset;
814 return true;
815 }
816
817 return false;

880 inline ts_ms_t shift_nonexistent_local_time_ms(ts_ms_t local_ms,
881 TimeZone zone,
882 int direction) {
883 const ts_ms_t window = static_cast<ts_ms_t>(MS_PER_DAY) * 2;
884 const bool forward = direction >= 0;
885 ts_ms_t low = forward ? local_ms : local_ms - window;
886 ts_ms_t high = forward ? local_ms + window : local_ms;
887
888 LocalTimeResolution edge =
889 resolve_local_time_ms(forward ? high : low, zone);
890 if(!local_time_status_has_utc(edge.status)) {
891 return ERROR_TIMESTAMP;
892 }
893
894 while(high - low > 1) {
895 const ts_ms_t mid = low + (high - low) / 2;
896 const LocalTimeResolution resolution =
897 resolve_local_time_ms(mid, zone);
898 if(local_time_status_has_utc(resolution.status)) {
899 if(forward) {
900 high = mid;
901 } else {
902 low = mid;
903 }
904 } else if(forward) {
905 low = mid;
906 } else {
907 high = mid;
908 }
909 }
910
911 const LocalTimeResolution shifted =
912 resolve_local_time_ms(forward ? high : low, zone);
913 return local_time_resolution_to_utc(
914 shifted,
915 AmbiguousTimePolicy::first_occurrence);
916 }
