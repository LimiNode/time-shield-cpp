31 TIME_SHIELD_CONSTEXPR inline bool oadate_can_cast_to_i64(oadate_t value) noexcept {
32 return value >= static_cast<oadate_t>((std::numeric_limits<int64_t>::min)())
33 && value <= static_cast<oadate_t>((std::numeric_limits<int64_t>::max)());

36 TIME_SHIELD_CONSTEXPR inline oadate_t oadate_abs(oadate_t value) noexcept {
37 return value < 0 ? -value : value;

40 TIME_SHIELD_CONSTEXPR inline oadate_t oadate_trunc_toward_zero(oadate_t value) noexcept {
41 if (!oadate_can_cast_to_i64(value)) {
42 return value;
43 }
44 return static_cast<oadate_t>(static_cast<int64_t>(value));
45 }

47 TIME_SHIELD_CONSTEXPR inline oadate_t oadate_floor_value(oadate_t value) noexcept {
48 const oadate_t truncated = oadate_trunc_toward_zero(value);
49 if (truncated == value) {
50 return truncated;
51 }
52 if (value < 0) {
53 return truncated - static_cast<oadate_t>(1.0);
54 }
55 return truncated;
56 }

58 TIME_SHIELD_CONSTEXPR inline bool oadate_has_fraction(oadate_t value) noexcept {
59 if (!oadate_can_cast_to_i64(value)) {
60 return false;
61 }
62 return oadate_trunc_toward_zero(value) != value;
63 }

65 TIME_SHIELD_CONSTEXPR inline oadate_t linear_days_to_oadate(oadate_t linear_days) noexcept {
66 if (linear_days < 0 && oadate_has_fraction(linear_days)) {
67 const oadate_t whole_days = oadate_floor_value(linear_days);
68 const oadate_t fraction = linear_days - whole_days;
69 return whole_days - fraction;
70 }
71 return linear_days;
72 }
73
74 TIME_SHIELD_CONSTEXPR inline oadate_t oadate_to_linear_days(oadate_t oa) noexcept {
75 if (oa < 0 && oadate_has_fraction(oa)) {
76 const oadate_t whole_days = oadate_trunc_toward_zero(oa);
77 const oadate_t fraction = oadate_abs(oa - whole_days);
78 return whole_days + fraction;
79 }
80 return oa;
81 }
