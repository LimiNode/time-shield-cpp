136 bool is_first = m_measure_immediately;
137 while (!m_is_stop_requested.load()) {

222 bool init(std::chrono::milliseconds interval, bool measure_immediately = true) {
223 if (is_process_shutting_down()) {
224 return false;
225 }
226
227 std::unique_ptr<RunnerT> local_runner;
228 std::chrono::milliseconds start_interval = interval;
229 bool use_measure_immediately = measure_immediately;
230 {
231 std::unique_lock<std::mutex> lk(m_mtx);
232 while (is_transitioning_locked()) {
233 m_cv.wait(lk);
234 }
235 if (is_process_shutting_down_locked()) {
236 return false;
237 }
238 if (m_state == State::running && is_running_locked()) {
239 return true;
240 }
241 if (start_interval.count() <= 0) {
242 start_interval = std::chrono::milliseconds(1);
243 }
244 m_interval = start_interval;
245 m_measure_immediately = use_measure_immediately;
246 m_state = State::starting;
247
248 local_runner = build_runner_locked();
249 if (!local_runner) {
250 m_state = State::stopped;
251 lk.unlock();
252 m_cv.notify_all();
253 return false;
254 }
255 }
256
257 bool has_started = false;
258 bool is_ok = false;
259 try {
260 has_started = local_runner->start(start_interval, use_measure_immediately);
261 if (has_started) {
262 is_ok = local_runner->measure_now();
263 }
264 } catch (...) {
265 is_ok = false;
266 }
267
268 if (!has_started || !is_ok || is_process_shutting_down()) {
269 try {
270 if (has_started) {
271 local_runner->stop();
272 }
273 } catch (...) {
274 // no-throw
275 }
276 local_runner.reset();
277 }
278
279 {
280 std::lock_guard<std::mutex> lk(m_mtx);
281 if (local_runner) {
282 m_last_offset_us.store(local_runner->offset_us(), std::memory_order_relaxed);
283 m_runner = std::move(local_runner);
284 m_state = State::running;
285 } else {
286 m_runner.reset();
287 m_state = State::stopped;
288 }
289 }
290 m_cv.notify_all();
291 return is_ok;
292 }
